AUTH HELPER (email OTP -> token; devOtp is echoed in dev)
  API=http://localhost:5001/api/v1
  H='-H Content-Type:application/json -H X-App-Source:flutter_android'
  login(){ R=$(curl -s $H -X POST "$API/auth/email/otp/send" -d "{\"email\":\"$1\"}");
    SID=$(echo "$R"|sed -n 's/.*"sessionId":"\([^"]*\)".*/\1/p');
    OTP=$(echo "$R"|sed -n 's/.*"devOtp":"\([^"]*\)".*/\1/p');
    curl -s $H -X POST "$API/auth/email/otp/verify" \
      -d "{\"sessionId\":\"$SID\",\"otp\":\"$OTP\",\"deviceInfo\":\"qa\"}" \
      | sed -n 's/.*"accessToken":"\([^"]*\)".*/\1/p'; }
  TOK=$(login qa1@example.com)   # then: curl $H -H "Authorization: Bearer $TOK" ...
